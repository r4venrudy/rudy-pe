cmake_minimum_required(VERSION 3.20)
project(rudy_pe VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(RUDY_PE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
option(RUDY_PE_BUILD_FUZZER "Build the optional Clang/libFuzzer target" OFF)

add_library(rudy_pe_lib src/pe.cpp)
add_library(rudy_pe::lib ALIAS rudy_pe_lib)
target_include_directories(rudy_pe_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rudy_pe_lib PUBLIC cxx_std_20)
set_target_properties(rudy_pe_lib PROPERTIES CXX_EXTENSIONS OFF)

add_executable(rudy-pe src/main.cpp)
target_link_libraries(rudy-pe PRIVATE rudy_pe::lib)

if(MSVC)
    target_compile_options(rudy_pe_lib PRIVATE /W4 /permissive-)
    target_compile_options(rudy-pe PRIVATE /W4 /permissive-)
    if(RUDY_PE_WARNINGS_AS_ERRORS)
        target_compile_options(rudy_pe_lib PRIVATE /WX)
        target_compile_options(rudy-pe PRIVATE /WX)
    endif()
else()
    target_compile_options(rudy_pe_lib PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow)
    target_compile_options(rudy-pe PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow)
    if(RUDY_PE_WARNINGS_AS_ERRORS)
        target_compile_options(rudy_pe_lib PRIVATE -Werror)
        target_compile_options(rudy-pe PRIVATE -Werror)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(rudy-pe-tests tests/test_pe.cpp)
    target_link_libraries(rudy-pe-tests PRIVATE rudy_pe::lib)
    if(MSVC)
        target_compile_options(rudy-pe-tests PRIVATE /W4 /permissive-)
        if(RUDY_PE_WARNINGS_AS_ERRORS)
            target_compile_options(rudy-pe-tests PRIVATE /WX)
        endif()
    else()
        target_compile_options(rudy-pe-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow)
        if(RUDY_PE_WARNINGS_AS_ERRORS)
            target_compile_options(rudy-pe-tests PRIVATE -Werror)
        endif()
    endif()
    add_test(NAME parser-unit COMMAND rudy-pe-tests)

    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli-integration
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:rudy-pe>)
endif()

if(RUDY_PE_BUILD_FUZZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "RUDY_PE_BUILD_FUZZER requires Clang with libFuzzer support")
    endif()
    add_executable(rudy-pe-fuzzer tests/fuzz_pe.cpp)
    target_link_libraries(rudy-pe-fuzzer PRIVATE rudy_pe::lib)
    target_compile_options(rudy_pe_lib PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
    target_compile_options(rudy-pe-fuzzer PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(rudy-pe-fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

install(TARGETS rudy-pe RUNTIME DESTINATION bin)
